# Client dependencies for the kordinate CLI (installer/kordinate-cli).
# The workstation image has its own dependencies in the Dockerfile.
#
# Command         Needs
# -------         -----
# init, join      curl, ssh, python3, python3-yaml
# connect         ssh, curl, cloudflared (auto-installed)
#
# Tool          Package              Notes
# ----          -------              -----
curl            # apt install curl
ssh             # apt install openssh-client
python3         # apt install python3               (init/join only)
python3-yaml    # apt install python3-yaml           (init/join only)
cloudflared     # auto-installed by 'kordinate connect'
